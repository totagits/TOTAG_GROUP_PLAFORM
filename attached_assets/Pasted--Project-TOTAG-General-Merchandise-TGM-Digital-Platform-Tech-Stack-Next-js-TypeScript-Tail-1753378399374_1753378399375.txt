// prisma/schema.prisma

// Define your database schema for inventory, orders, suppliers, logistics

datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

generator client {
  provider = "prisma-client-js"
}

model Inventory {
  id          Int      @id @default(autoincrement())
  sku         String   @unique
  productName String
  quantity    Int      @default(0)
  price       Float
  updatedAt   DateTime @updatedAt
}

model Order {
  id           Int      @id @default(autoincrement())
  inventory    Inventory @relation(fields: [inventoryId], references: [id])
  inventoryId  Int
  quantity     Int
  customerName String
  status       String   @default("Pending")
  createdAt    DateTime @default(now())
}

model Supplier {
  id           Int      @id @default(autoincrement())
  name         String
  contactEmail String   @unique
  phone        String
  address      String
  createdAt    DateTime @default(now())
}

model Logistics {
  id           Int      @id @default(autoincrement())
  order        Order    @relation(fields: [orderId], references: [id])
  orderId      Int      @unique
  latitude     Float
  longitude    Float
  updatedAt    DateTime @updatedAt
}

// components/InventoryTable.tsx
import { useQuery } from '@tanstack/react-query';
import axios from 'axios';
export default function InventoryTable() {
  const { data, isLoading } = useQuery(['inventory'], () => axios.get('/api/inventory').then(res => res.data));
  if(isLoading) return <p>Loading...</p>;
  return (
    <table className="min-w-full bg-white">
      <thead><tr><th>SKU</th><th>Product</th><th>Qty</th><th>Price</th><th>Updated</th></tr></thead>
      <tbody>
        {data.map((item: any) => (
          <tr key={item.id} className="border-t">
            <td className="p-2">{item.sku}</td>
            <td className="p-2">{item.productName}</td>
            <td className="p-2">{item.quantity}</td>
            <td className="p-2">${item.price.toFixed(2)}</td>
            <td className="p-2">{new Date(item.updatedAt).toLocaleDateString()}</td>
          </tr>
        ))}
      </tbody>
    </table>
  );
}

// components/OrderForm.tsx
import { useQueryClient, useMutation } from '@tanstack/react-query';
import axios from 'axios';
import { useState } from 'react';
export default function OrderForm() {
  const qc = useQueryClient();
  const { data: inventory } = useQuery(['inventory'], () => axios.get('/api/inventory').then(res=>res.data));
  const [form, setForm] = useState({ inventoryId: '', quantity: '', customerName: '' });
  const mutation = useMutation((newOrder: any) => axios.post('/api/orders', newOrder), {
    onSuccess: () => qc.invalidateQueries(['orders'])
  });
  const handleSubmit = (e: any) => { e.preventDefault(); mutation.mutate({ ...form, quantity: Number(form.quantity), inventoryId: Number(form.inventoryId) }); };
  return (
    <form onSubmit={handleSubmit} className="space-y-4">
      <select name="inventoryId" required onChange={e=>setForm({...form, [e.target.name]: e.target.value})} className="border p-2 rounded">
        <option value="">Select Product</option>
        {inventory?.map((i:any)=><option key={i.id} value={i.id}>{i.productName}</option>)}
      </select>
      <input type="number" name="quantity" placeholder="Quantity" required onChange={e=>setForm({...form,[e.target.name]:e.target.value})} className="border p-2 rounded w-full" />
      <input type="text" name="customerName" placeholder="Customer Name" required onChange={e=>setForm({...form,[e.target.name]:e.target.value})} className="border p-2 rounded w-full" />
      <button type="submit" className="px-4 py-2 bg-blue-600 text-white rounded">Place Order</button>
    </form>
  );
}

// components/LogisticsMap.tsx
import { MapContainer, TileLayer, Marker, Popup } from 'react-leaflet';
import { useQuery } from '@tanstack/react-query';
import axios from 'axios';
export default function LogisticsMap() {
  const { data, isLoading } = useQuery(['logistics'], () => axios.get('/api/logistics').then(res=>res.data));
  if(isLoading) return <p>Loading map...</p>;
  return (
    <MapContainer center={[6.3, -10.8]} zoom={7} className="h-96 w-full">
      <TileLayer url="https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png" />
      {data.map((loc:any)=><Marker key={loc.id} position={[loc.latitude, loc.longitude]}> <Popup>Order {loc.orderId}</Popup></Marker>)}
    </MapContainer>
  );
}

// components/DashboardSummary.tsx
import { useQuery } from '@tanstack/react-query';
import axios from 'axios';
export default function DashboardSummary() {
  const inv = useQuery(['inventory'], () => axios.get('/api/inventory').then(res=>res.data));
  const ord = useQuery(['orders'], () => axios.get('/api/orders').then(res=>res.data));
  const sup = useQuery(['suppliers'], () => axios.get('/api/suppliers').then(res=>res.data));
  if(inv.isLoading||ord.isLoading||sup.isLoading) return <p>Loading...</p>;
  return (
    <div className="grid grid-cols-1 md:grid-cols-3 gap-6">
      <div className="p-6 bg-white rounded shadow"><h2 className="text-lg">Products</h2><p className="text-2xl">{inv.data.length}</p></div>
      <div className="p-6 bg-white rounded shadow"><h2 className="text-lg">Orders</h2><p className="text-2xl">{ord.data.length}</p></div>
      <div className="p-6 bg-white rounded shadow"><h2 className="text-lg">Suppliers</h2><p className="text-2xl">{sup.data.length}</p></div>
    </div>
  );
}
