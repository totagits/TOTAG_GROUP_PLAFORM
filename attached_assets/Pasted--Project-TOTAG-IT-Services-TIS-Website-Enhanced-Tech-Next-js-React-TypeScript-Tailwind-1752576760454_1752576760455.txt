// components/ServiceCard.tsx
import { motion } from 'framer-motion';
import Image from 'next/image';
export interface ServiceCardProps { title: string; description: string; href: string; iconSrc: string; }
export default function ServiceCard({ title, description, href, iconSrc }: ServiceCardProps) {
  return (
    <motion.div className="p-6 bg-white rounded-lg shadow hover:shadow-md transition" whileHover={{ scale: 1.02 }}>
      <Image src={iconSrc} alt={`${title} icon`} width={48} height={48} />
      <h3 className="text-xl font-semibold mb-2 mt-4">{title}</h3>
      <p className="text-gray-600 mb-4">{description}</p>
      <a href={href} className="text-brand-700 hover:underline">Learn More →</a>
    </motion.div>
  );
}

// Example test: __tests__/ServiceCard.test.tsx
import { render, screen } from '@testing-library/react';
import ServiceCard from '../components/ServiceCard';

test('renders ServiceCard with title and description', () => {
  render(<ServiceCard title="Test Service" description="Test Description" href="/test" iconSrc="/test-icon.png" />);
  expect(screen.getByText('Test Service')).toBeInTheDocument();
  expect(screen.getByText('Test Description')).toBeInTheDocument();
});
