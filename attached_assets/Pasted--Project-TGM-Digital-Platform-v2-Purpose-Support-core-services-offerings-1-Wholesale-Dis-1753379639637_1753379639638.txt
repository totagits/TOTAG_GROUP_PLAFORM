// prisma/schema.prisma

datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

generator client {
  provider = "prisma-client-js"
}

model Inventory {
  id          Int      @id @default(autoincrement())
  sku         String   @unique
  name        String
  quantity    Int      @default(0)
  price       Float
  updatedAt   DateTime @updatedAt
  wholesale   WholesaleOrder[]
  retail      RetailSale[]
}

model WholesaleOrder {
  id           Int      @id @default(autoincrement())
  inventory    Inventory @relation(fields: [inventoryId], references: [id])
  inventoryId  Int
  quantity     Int
  partner      String
  createdAt    DateTime @default(now())
}

model RetailSale {
  id           Int      @id @default(autoincrement())
  inventory    Inventory @relation(fields: [inventoryId], references: [id])
  inventoryId  Int
  quantity     Int
  outlet       String
  saleDate     DateTime @default(now())
}

model Supplier {
  id           Int      @id @default(autoincrement())
  name         String
  email        String   @unique
  phone        String
  address      String
  createdAt    DateTime @default(now())
}

model Logistics {
  id           Int      @id @default(autoincrement())
  orderType    String   // "wholesale" or "retail"
  orderId      Int
  lat          Float
  lng          Float
  updatedAt    DateTime @updatedAt
}

// components/InventoryTable.tsx
import { useQuery } from '@tanstack/react-query'; import axios from 'axios';
export default function InventoryTable(){ const {data,isLoading}=useQuery(['inv'],()=>axios.get('/api/inventory').then(r=>r.data)); if(isLoading)return<>'Loading'</>;
  return(<table className="w-full bg-white"><thead><tr><th>SKU</th><th>Name</th><th>Qty</th><th>Price</th><th>Updated</th></tr></thead><tbody>{data.map((i:any)=><tr key={i.id}><td>{i.sku}</td><td>{i.name}</td><td>{i.quantity}</td><td>${i.price.toFixed(2)}</td><td>{new Date(i.updatedAt).toLocaleDateString()}</td></tr>)}</tbody></table>);
}

// components/WholesaleOrderForm.tsx
import { useState } from 'react'; import { useQuery,useMutation,useQueryClient } from '@tanstack/react-query'; import axios from 'axios';
export default function WholesaleOrderForm(){ const qc=useQueryClient(); const {data:inv}=useQuery(['inv'],()=>axios.get('/api/inventory').then(r=>r.data)); const [f,setF]=useState({inventoryId:'',quantity:'',partner:''}); const mut=useMutation((d:any)=>axios.post('/api/wholesale',d),{onSuccess:()=>qc.invalidateQueries(['wholesale'])});
  return(<form onSubmit={e=>{e.preventDefault(); mut.mutate({inventoryId:+f.inventoryId,quantity:+f.quantity,partner:f.partner});}} className="space-y-2 p-4 bg-white rounded">
    <select name="inventoryId" onChange={e=>setF({...f,[e.target.name]:e.target.value})} required className="border p-2 rounded w-full"><option value="">Product</option>{inv?.map((i:any)=><option key={i.id} value={i.id}>{i.name}</option>)}</select>
    <input type="number" name="quantity" placeholder="Quantity" onChange={e=>setF({...f,[e.target.name]:e.target.value})} required className="border p-2 rounded w-full"/>
    <input type="text" name="partner" placeholder="Partner Name" onChange={e=>setF({...f,[e.target.name]:e.target.value})} required className="border p-2 rounded w-full"/>
    <button type="submit" className="bg-blue-600 text-white px-4 py-2 rounded">Place Wholesale Order</button>
  </form>);
}
